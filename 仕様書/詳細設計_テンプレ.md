```csharp
[Test]
public void TakeDamage_ValidDamage_ReducesHealth()
```

### ログ出力例
```csharp
/// <summary>
/// カスタムログメソッド
/// </summary>
/// <param name="message">メッセージ</param>
/// <param name="level">ログレベル</param>
[System.Diagnostics.Conditional("UNITY_EDITOR")]
private void LogDebug(string message, LogLevel level = LogLevel.Debug)
{
    string formattedMessage = $"[{GetType().Name}] {message}";
    
    switch (level)
    {
        case LogLevel.Debug:
            Debug.Log(formattedMessage);
            break;
        case LogLevel.Warning:
            Debug.LogWarning(formattedMessage);
            break;
        case LogLevel.Error:
            Debug.LogError(formattedMessage);
            break;
    }
}
```

### 例外処理
```csharp
public bool TryExecuteAction(System.Action action)
{
    try
    {
        action?.Invoke();
        return true;
    }
    catch (System.Exception ex)
    {
        LogDebug($"Action execution failed: {ex.Message}", LogLevel.Error);
        return false;
    }
}
```
